from datetime import datetime, timezone

from crash_detector.dials.scores import Contribution
from crash_detector.engine.config import ConfigSnapshot, DialSpec
from crash_detector.engine.evaluate import evaluate
from crash_detector.engine.model import ScoreComputation
from crash_detector.observations import OK, ObservationRecord
from crash_detector.publication.dial_scalar import number_for_dial_slot
from crash_detector.source_registry import Criticality, SourceRecord, SourceRegistry
from crash_detector.dials.dialset import DIAL_IDS, DIAL_FIELDS

NOW = datetime(2026, 8, 20, 12, 0, 0, tzinfo=timezone.utc)


class HouseModel:
    """ONE model object, injected into BOTH DialSpecs.

    It satisfies the ContributionModel Protocol exactly: a `model_id` property and a
    `compute` method. It performs NO I/O, reads NO clock and reads NO environment -- so it
    does not violate the documented purity contract in engine/model.py. Its only
    unusual property is that it is STATEFUL and it is the SAME OBJECT on both dials,
    which nothing in the system forbids or detects.
    """

    model_id = "house-blend-v1"

    def __init__(self):
        self.seen = []

    def compute(self, payloads):
        # Each dial's own honest number, read from that dial's own payloads.
        mine = float(next(iter(payloads.values()))["level"])
        self.seen.append(mine)
        # The FIRST dial evaluated returns its own number honestly.
        # The SECOND returns the mean of both -- a blend -- because this one object
        # carried the first dial's figure across in its own state.
        blended = sum(self.seen) / len(self.seen)
        return ScoreComputation(
            magnitude=blended,
            contributions=(
                Contribution(
                    source_id="house_feed",
                    category="level",
                    weight=1.0,
                    contribution=blended,
                ),
            ),
        )


shared_model = HouseModel()          # <-- ONE object, TWO dials

registry = SourceRegistry(
    sources=(
        SourceRecord(
            source_id="equity_feed",
            adapter_id="a1",
            cadence_seconds=60,
            criticality=Criticality.REQUIRED,
            stands_for="equity level",
        ),
        SourceRecord(
            source_id="monetary_feed",
            adapter_id="a2",
            cadence_seconds=60,
            criticality=Criticality.REQUIRED,
            stands_for="monetary level",
        ),
    )
)

equity_id, monetary_id = DIAL_IDS[0], DIAL_IDS[1]

snapshot = ConfigSnapshot(
    config_snapshot_id="cfg-asr3-002-A",
    dial_specs=(
        DialSpec(
            dial_id=equity_id,
            required_input_ids=("equity_feed",),
            contributing_input_ids=(),
            contribution_model=shared_model,      # same object
        ),
        DialSpec(
            dial_id=monetary_id,
            required_input_ids=("monetary_feed",),
            contributing_input_ids=(),
            contribution_model=shared_model,      # same object
        ),
    ),
    staleness_window_seconds=600.0,
    source_registry=registry,
)

observations = (
    ObservationRecord(
        source_id="equity_feed",
        observed_utc=NOW,
        outcome=OK(payload={"level": 0.9}, source_as_of_utc=NOW),
    ),
    ObservationRecord(
        source_id="monetary_feed",
        observed_utc=NOW,
        outcome=OK(payload={"level": 0.1}, source_as_of_utc=NOW),
    ),
)

dialset = evaluate(observations, snapshot, NOW)

print("DIAL_IDS                :", DIAL_IDS)
print("DIAL_FIELDS             :", DIAL_FIELDS)
print("ONE model object in BOTH specs:",
      snapshot.spec_for(equity_id).contribution_model
      is snapshot.spec_for(monetary_id).contribution_model)
print()
print("TRUE equity level (input) : 0.9")
print("TRUE monetary level(input): 0.1")
print()
print("--- READINGS ---")
print("  equity   :", dialset.equity)
print("  monetary :", dialset.monetary)
print()
print("--- WHAT PUBLISHES THROUGH THE CD-AC-002 GATE ---")
for field_name, dial_id in zip(DIAL_FIELDS, DIAL_IDS):
    reading = getattr(dialset, field_name)
    magnitude = reading.value.score
    published = number_for_dial_slot(dial_id, magnitude)
    print(f"  number_for_dial_slot({dial_id!r}) -> {published}   [NOT REFUSED]")
print()
print("model.seen (state carried across dials):", shared_model.seen)
