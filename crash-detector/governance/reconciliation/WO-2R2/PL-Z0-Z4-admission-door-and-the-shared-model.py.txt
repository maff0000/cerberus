def magnitude(dial_id, quantity, sources):
    return DialMagnitude(
        dial_id=dial_id,
        quantity=observed_for_dial(
            dial_id=dial_id, quantity=quantity, inputs_used=tuple(sources)

    )

# --- PL-Z4  the seat's declared OPEN route, written independently by the PL --------------
class PLCarryingModel:
    """Stateful across dials. The PL's own, not the seat's CarryingModel."""

    def __init__(self):
        self.seen = []

    @property
    def model_id(self):
        return "pl.z4.carrying.v0"

    def compute(self, payloads):
        figure = float(list(payloads.values())[0]["figure"])
        self.seen.append(figure)
        carried = sum(self.seen) / len(self.seen)
        return ScoreComputation(
            magnitude=carried,
            contributions=tuple(
                Contribution(source_id=s, category="carried", weight=1.0, contribution=0.1)
                for s in sorted(payloads)
            ),
        )


def one_model_both_dials():
    model = PLCarryingModel()
    equity = DialSpec(
        dial_id="equity_crash",
        required_input_ids=("vix",),
        contributing_input_ids=(),

    )
    monetary = DialSpec(
        dial_id="monetary_debasement",
        required_input_ids=("dxy",),
        contributing_input_ids=(),
        contribution_model=model,
    )
    registry = SourceRegistry(
        sources=(

                         stands_for="PL PROBE double; no ingredient is registered in WO-2"),
            SourceRecord(source_id="dxy", adapter_id="fake.adapter.v0", cadence_seconds=60,
                         criticality=Criticality.REQUIRED,
                         stands_for="PL PROBE double; no ingredient is registered in WO-2"),
        )
    )
    snapshot = ConfigSnapshot(
        config_snapshot_id="cfg-pl-z4",
        dial_specs=(equity, monetary),
        staleness_window_seconds=600.0,
        source_registry=registry,
    )
    observations = (
        ObservationRecord(source_id="vix", observed_utc=NOW,
                          outcome=OK(payload={"figure": 0.9},
                                     source_as_of_utc=NOW - timedelta(seconds=10))),
        ObservationRecord(source_id="dxy", observed_utc=NOW,
                          outcome=OK(payload={"figure": 0.1},

    )
    dials = evaluate(observations, snapshot, NOW)
    print(f"  equity   : {dials.equity!r}")
    print(f"  monetary : {dials.monetary!r}")
    return dials
